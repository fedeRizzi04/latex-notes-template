% Stile per lo pseudocodice: mathescape=true permette di scrivere $...$
% dentro l'ambiente e viene reso come formula. Nessun comando necessario:
% basta scrivere il testo, es.
%   while $i \neq 0$ do
%       a $\gets$ 5
\lstdefinestyle{pseudo}{
    basicstyle=\ttfamily\small,
    mathescape=true,
    escapechar=\%,                 % usa %...% per commenti in corsivo, es: %questo è un commento%
    tabsize=2,
    columns=fullflexible,
    keepspaces=true,
    showstringspaces=false,
    breaklines=true,
    breakindent=1.5em,
    numbers=left,
    numberstyle=\tiny\color{secondario},
    numbersep=10pt,
    xleftmargin=1.8em,
    keywordstyle=\bfseries,         % nero, niente colore: stile "libro serio"
    commentstyle=\itshape\color{secondario},
    % parole chiave riconosciute automaticamente (italiano + inglese)
    keywords={%
        while,do,end,if,then,else,for,to,downto,repeat,until,%
        return,function,procedure,begin,and,or,not,in,break,continue,%
        mentre,fare,fine,se,allora,altrimenti,per,ripeti,finche,%
        restituisci,funzione,procedura,inizio,e,o,non%
    },
}
\lstset{style=pseudo}

% Ambiente "algoritmo": NON è un box colorato
% Si usa esattamente come prima:
%
%   \begin{algoritmo}{Titolo dell'algoritmo}
%   procedure Esempio(n)
%       i $\gets$ 0
%       while $i \neq n$ do
%           i $\gets$ i + 1
%       return i
%   \end{algoritmo}
%
\NewTCBListing{algoritmo}{O{} g}{%
    enhanced,
    breakable,
    listing only,
    listing options={style=pseudo},
    colback=white,          % nessuno sfondo
    colbacktitle=white,     % titolo su sfondo bianco, non su barra colorata
    colframe=black,
    coltitle=black,
    fonttitle=\bfseries,
    boxrule=0pt,             % nessun bordo
    leftrule=0pt,rightrule=0pt,
    toprule=0.9pt,           % riga sottile sopra (sotto la didascalia)
    bottomrule=0.4pt,        % riga sottile sotto il codice
    arc=0pt,
    left=0pt,right=0pt,top=3pt,bottom=3pt,
    title={\titoloambiente{\txtalgoritmo~\thetcbcounter}{#2}},
    #1
}

\makeatletter
\renewenvironment{proof}[1][\txtdimostrazione]{\par\vspace{-0.35\baselineskip}
    \pushQED{\qed}%
    \normalfont \topsep2\p@\@plus2\p@\relax
    \noindent\textcolor{secondario}{\rule{\linewidth}{0.4pt}}\par
    \trivlist
    \item[\hskip\labelsep\bfseries #1\@addpunct{.}]\ignorespaces
}{%
    \popQED\endtrivlist\@endpefalse
    \par\vspace{1pt}\noindent\textcolor{secondario}{\rule{\linewidth}{0.4pt}}\par
}
\makeatother
